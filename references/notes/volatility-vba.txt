Function BS_price(S, K, T, r, vol, IsCall, n)
'n=0 returns price, n=1 returns d1, n=2 returns d2
Dim d1, d2 As Double

d1 = (Log(S / K) + (r + vol ^ 2 / 2) * T) / (vol * T ^ (1 / 2))
d2 = d1 - vol * T ^ (1 / 2)

N_d1 = Application.WorksheetFunction.Norm_S_Dist(d1, True)
N_d2 = Application.WorksheetFunction.Norm_S_Dist(d2, True)
N_minusd1 = Application.WorksheetFunction.Norm_S_Dist(-d1, True)
N_minusd2 = Application.WorksheetFunction.Norm_S_Dist(-d2, True)


If IsCall = 1 Then 'if iscall = 1 it's a call option
    BS_price = S * N_d1 - K * N_d2 * Exp(-r * T)

Else 'it's a put
    BS_price = K * N_minusd2 * Exp(-r * T) - S * N_minusd1
    
End If


End Function


Function impliedvol(S, K, T, r, price As Double, IsCall As Integer)

Dim vol As Double 'testing volatility
Dim i As Integer
vol = 0.22 'start value for the volatility
test_price = BS_price(S, K, T, r, vol, IsCall, 0) 'BS price using the testing vol
vega = S * T ^ (1 / 2) * Application.WorksheetFunction.Norm_S_Dist(d1, True) ' vega of the option
i = 0

Do
vol = vol + (price - test_price) / vega
    If vol < 0.00001 Then
    vol = 0.00001
    End If
d1 = BS_price(S, K, T, r, vol, IsCall, 1) 'd1 from BS
vega = S * T ^ (1 / 2) * Application.WorksheetFunction.Norm_S_Dist(d1, True)

End Function
